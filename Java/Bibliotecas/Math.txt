Math.max(x, y); // Retorna o maior número
Math.min(x, y); // Retorna o menor número
Math.abs(x); // Retorna o número positivo

Math.sqrt(x); // Retorna a raíz quadrada do número
Math.log(x); // Retorna o logaritmo natural do número
Math.log10(x); // Retorna o logaritmo na base 10 do número

Math.round(x); // Arredonda o número
Math.ceil(x); // Arredonda o número pra cima
Math.floor(x); // Arredonda o número pra baixo
