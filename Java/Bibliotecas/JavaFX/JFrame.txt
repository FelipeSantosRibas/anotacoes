Um JFrame é uma janela de GUI.

------------------------------------------------------------------

import javax.swing.Jframe; // Importa a biblioteca JFrame;

------------------------------------------------------------------

JFrame frame = new JFrame(); // Cria um novo frame
frame.setTitle("Título"); // Define o título do frame
frame.setIconImage(imagem.getImage()); // Define a imagem do ícone do frame (Use a biblioteca ImageIcon!!!)
frame.getContentPane().setBackground(cor);  // Define a cor de fundo do frame (Use a biblioteca Color!!!)
frame.setDefaultCloseOperation(); // Define o que acontece ao fechar o frame
frame.setResizable(boolean); // Define se o frame é redimensionável
frame.setSize(width, height); // Define o tamanho do frame
frame.setMinimumSize(Dimension); // Define o tamanho mínimo do frame (use a biblioteca Dimension !!!)
frame.setMaximumSize(Dimension); // Define o tamanho mínimo do frame (use a biblioteca Dimension !!!)
frame.setVisible(boolean); // Define se o frame é visível
frame.add() // Adiciona algo, como JLabels ou JPanels


/* IMPORTANTE: Existem diversar maneiras de se formatar um JFrame, use de acordo com o que mais combina com sua necessidade atual, alguns jeitos úteis são: usar layout padrão, layout null, pack(), etc.*/

frame.setLayout(null); // Define o layout do frame, pode ser null, new BorderLayout()
frame.pack(); // Ajusta o tamanho do frame para acomodar todos os componentes atuais
