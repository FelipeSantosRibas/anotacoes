Um JButton é um botão que performa ações quando clicado.

------------------------------------------------------------------

import javax.swing.JButton; // Importa a biblioteca JButton

------------------------------------------------------------------

// Botão
JButton button = new JButton(); // Cria um novo JButton
button.setBounds(x,y,width,height); // Define o tamanho e a posição do JButton
button.setEnabled(false); // Define se o botão está ativo

// Aparência
button.setBackground(corDeFundo); // Define a cor de fundo do JButton
button.setOpaque(true); // Define se o JButton é opaco
button.setText("texto"); // Define o texto do botão
button.setFocusable(); // Define se o botão é "focável" (deixe false para evitar bordas ao redor do texto)

// Ação
button.addActionListener(e -> ação); // Executa uma ação quando clicado
button.addActionListener(this); // Executa uma ação quando clicado (Use a biblioteca ActionListener!!!)
