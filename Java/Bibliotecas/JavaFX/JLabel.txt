Um JLabel é uma área de display de GUI para strings e imagens.

------------------------------------------------------------------

import javax.swing.JLabel; // Importa a biblioteca JLabel

------------------------------------------------------------------

JLabel label = new JLabel(); // Cria um novo JLabel

// Label
label.setBackground(cor); // Define a cor de fundo do JLabel(precisa ser opaco)
label.setOpaque(true); // Define se o JLabel é opaco
label.setBorder(border); // Define a borda do JLabel (Use a biblioteca Border e BorderFactory!!!)
this.setBounds(x, y, width, height); // Define a posição e o tamanho do JLabel (Layout do JFrame precisa ser null!!!)


// Formato do texto
label.setText("Texto"); // Define o texto do JLabel
label.setForeground(cor); // Define a cor do texto (Use a biblioteca Color!!!)
label.setFont(fonte)); // Define a fonte do texto (Use a biblioteca Font!!!)

// Posição do texto
label.setHorizontalTextPosition(JLabel.POSITION) // Define a posição horizontal do texto em relação ao ImageIcon
label.setVerticalTextPosition(JLabel.POSITION); // Define a posição vertical do texto em relação ao ImageIcon
label.setIconTextGap(-50); // Define a distância entre o texto e o ImageIcon


// ImageIcon
label.setIcon(imagemLabel); // Define a imagem do label
label.setVerticalAlignment(JLabel.POSITION); // Define a posição vertical do ImageIcon
label.setHorizontalAlignment(JLabel.POSITION); // Define o posição horizontal do ImageIcon
