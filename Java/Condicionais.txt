IF (se):

if ( <condição> ) {
	<instrução>;
}

---------------------------------------------------------

IF ELSE(se senão):

if ( <condição> ) {
	<instrução>;
} else{
	<instrução>;
}

---------------------------------------------------------

IF ELSE IF(se senão se):

if ( <condição> ) {
	<instrução>;
} else if ( <condição> ){
	<instrução>;
}

---------------------------------------------------------

SWITCH (Caso):

switch ( <expressão ou variável> ){
	case <valor 1>:
		<instrução 1>;
		break;
	case <valor 2>:
		<instrução 2>;
		break;
	default
		<instrução default>;
		break;
}

---------------------------------------------------------

SWITCH 2 (Caso):

switch ( <expressão ou variável> ){
	case <valor 1> -> <instrução 1>;
	case <valor 2> -> {<instrução 2>;}
	default -> <instrução default>;
}
